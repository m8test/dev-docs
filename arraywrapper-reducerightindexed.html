val words = arrayOf("hello", "world", "kotlin") // 从右到左连接所有单词，并添加索引信息 val joinedString = words.reduceRightIndexed { index, word, acc -> acc + index + "_ " + word } // 输出：连接后的字符串: 2_kotlin 1_world 0_hello _console.log("连接后的字符串: ", joinedString)
